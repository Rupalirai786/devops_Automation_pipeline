then inside git repository where pyhton and dockerfile is preset, created .gitrhub ->workflows -> ci-pipeline.yaml file .

name: automationscript
on:
 push:
  branches:
  - main
jobs:
 build-job:
  runs-on: ubuntu-latest
  steps:
   - name: cloning
     uses: actions/checkout@v4
   - name: generating image from dockerfile
     run: docker build -t img .
   - name: running container
     run: docker run --name cont img:latest
